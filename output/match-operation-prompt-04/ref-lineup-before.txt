export function RefereeLineups({ context }: { context: RefereeMatchContext }) {
  return <div className="rf-two-columns">{[context.home, context.away].map((side, index) => {
    const players = context.lineups.filter((lineup) => lineup.teamId === side.id);
    return <article className="rf-card" key={side.id ?? index}><header className="rf-card-header"><h2 className="rf-team-heading"><TeamMark team={side} />{side.name}</h2><span className="rf-count">{players.length} cầu thủ</span></header>{players.length ? <><div className="rf-lineup-status">{players.every((player) => player.confirmedAt) ? "✓ Đội hình đã được xác nhận" : "Đội hình chưa được xác nhận đầy đủ"}</div>{[true, false].map((starting) => <section key={String(starting)} className="rf-lineup-section"><h3>{starting ? "Đội hình xuất phát" : "Cầu thủ dự bị"}</h3><ul>{players.filter((player) => player.isStarting === starting).map((player) => <li key={player.id}><span className="rf-shirt">{player.jerseyNumber ?? "—"}</span><strong>{player.fullName}</strong><small>{player.position === "GK" ? "Thủ môn" : player.position === "DF" ? "Hậu vệ" : player.position === "MF" ? "Tiền vệ" : player.position === "FW" ? "Tiền đạo" : "Cầu thủ"}</small></li>)}</ul>{!players.some((player) => player.isStarting === starting) ? <p className="rf-muted">Chưa có cầu thủ {starting ? "xuất phát" : "dự bị"}.</p> : null}</section>)}</> : <EmptyState title="Chưa có đội hình">Đội bóng chưa gửi đội hình cho trận đấu này.</EmptyState>}</article>;
  })}</div>;
}
