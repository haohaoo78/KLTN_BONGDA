        <div className="rf-stack rf-live-tools"><article className="rf-card"><header className="rf-card-header"><h2>Công cụ tác nghiệp</h2></header><div className="rf-toolbox">
          {finalAuthority?clockActions({...context,clock:currentClock},serverNow).map(value=><button type="button" className="rf-button" disabled={disabled} key={value} onClick={()=>void clockAction(value)}>{clockActionLabels[value]}</button>):null}
          {canRecordEvents(context)?<><h3>Ghi nhận diễn biến</h3>{Object.entries(eventLabels).map(([value,label])=><button className={`rf-button rf-event-tool rf-tool-${value.toLowerCase()}`} disabled={disabled} type="button" key={value} onClick={()=>{const timing=currentEventTiming(context,elapsed);setEventType(value as MatchEvent["eventType"]);setEventTeam(context.home.id??"");setEventPlayer("");setEventTiming(timing);setEventMinute(String(timing.minute));setDescription("");}}>{label}</button>)}</>:<p className="rf-muted">{context.match.status!=="IN_PROGRESS"?"Công cụ ghi diễn biến chỉ mở khi trận đang thi đấu.":"Hiệp đấu, trạng thái đồng hồ hoặc nhiệm vụ chưa cho phép ghi diễn biến."}</p>}
        </div></article><RefereeCrewCard context={context}/></div>

      </div>
